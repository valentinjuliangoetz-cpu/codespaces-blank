<script>
let xp = parseInt(localStorage.getItem('xp')) || 0;

function updateDisplay() {
  const level = Math.floor(xp / 100) + 1;
  document.getElementById('xpDisplay').innerText = `XP: ${xp}`;
  document.getElementById('levelDisplay').innerText = `Level: ${level}`;

  const aura = document.getElementById('aura');
  const avatar = document.getElementById('avatar');

  // Change aura and avatar based on XP intensity
  if (xp < 200) {
    aura.style.background = "radial-gradient(circle, rgba(100,100,100,0.3), transparent 70%)";
    avatar.style.filter = "grayscale(70%) brightness(0.6)";
  } else if (xp < 1000) {
    aura.style.background = "radial-gradient(circle, rgba(255,215,0,0.5), transparent 70%)";
    avatar.style.filter = "grayscale(0%) brightness(1)";
  } else if (xp < 5000) {
    aura.style.background = "radial-gradient(circle, rgba(0,255,255,0.6), transparent 70%)";
    avatar.style.filter = "brightness(1.2)";
  } else {
    aura.style.background = "radial-gradient(circle, rgba(0,150,255,0.8), transparent 70%)";
    avatar.style.filter = "brightness(1.5) saturate(1.2)";
  }
}

function addXP(amount) {
  xp += amount;
  localStorage.setItem('xp', xp);
  updateDisplay();
}

document.querySelectorAll('input[type="checkbox"]').forEach(box => {
  box.addEventListener('change', (e) => {
    if (e.target.checked) {
      const gain = parseInt(e.target.getAttribute('data-xp'));
      addXP(gain);
      e.target.disabled = true; // prevent double XP for same task
    }
  });
});

updateDisplay();
</script>
